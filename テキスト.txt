function addMistake(q){
  var ex=mistakes.find(function(m){return m.q===q;});
  if(ex) ex.count++;
  else mistakes.push({q:q,count:1});
  updateUI();
}
function removeMistake(q){
  mistakes=mistakes.filter(function(m){return m.q!==q;});
  updateUI();
}

function shuf(a){var b=a.slice();for(var i=b.length-1;i>0;i--){var j=0|Math.random()*(i+1);var t=b[i];b[i]=b[j];b[j]=t;}return b;}

function startLevel(level){
  closeDiff();closeMenu();
  lv=level; isSudden=(level==='sudden'); suddenScore=0;
  var src;
  if(level==='sudden') src=shuf(DATA.slice());
  else src=DATA.filter(function(c){return c.t===level;});
  src=shuf(src);
  if(level==='easy'||level==='mid') src=src.slice(0,10);
  if(level==='hard') src=src.slice(0,10);
  deck=src; qi=0; okCnt=0; ngCnt=0; ep=0; sessionMisses=[];
  var modeNames={easy:'🌱 初級',mid:'🌿 中級',hard:'🔥 上級',sudden:'💀 サドンデス',review:'📖 弱点克服'};
  document.getElementById('q-mode').textContent=modeNames[level]||level;
  show('quiz-screen');
  renderQ();
}

function startReview(){
  if(mistakes.length===0) return;
  closeMenu();
  lv='review'; isSudden=false;
  var qqs=mistakes.map(function(m){return m.q;});

  qi=0; okCnt=0; ngCnt=0; ep=0; sessionMisses=[];
  document.getElementById('q-mode').textContent='📖 弱点克服';
  show('quiz-screen'); renderQ();
}

function replayLevel(){
  if(lv==='review') startReview();
  else startLevel(lv);
}

function startTimer(){
  stopTimer();
  if(!isSudden) return;
  document.getElementById('timer-container').style.display='block';
  timeLeft=TIME; updateTimerUI();
  timerIv=setInterval(function(){
    timeLeft-=.1;
    if(timeLeft<=0){timeLeft=0;updateTimerUI();clearInterval(timerIv);timeUp();}
    else updateTimerUI();
  },100);
}
function stopTimer(){clearInterval(timerIv);document.getElementById('timer-container').style.display='none';}
function updateTimerUI(){
  var pct=Math.max(0,timeLeft/TIME*100);
  var bar=document.getElementById('timer-bar');
  bar.style.width=pct+'%';
  bar.style.background=pct<25?'var(--accent-red)':pct<50?'var(--accent-gold)':'var(--accent-green)';
}
function timeUp(){
  if(qi>=deck.length) return;
  ngCnt++; addMistake(deck[qi].q);
  sessionMisses.push({q:deck[qi].q,type:'時間切れ'});
  document.getElementById('ja-choices').querySelectorAll('.c-btn').forEach(function(b){b.disabled=true;if(b.dataset.correct==='1')b.className='c-btn ok';});
  setFb(false,'⌛ 時間切れ！正解: '+deck[qi].a);
  document.getElementById('nxt-btn').style.display='block';
  setKen('error','時間切れだ！次は気をつけよう！');
  showResult(true,'⌛ タイムオーバーで終了！');
}

// 選択肢を生成して正誤判定するヘルパー
function makeChoices(gridId, opts, correctVal, onCorrect, onWrong){
  var g=document.getElementById(gridId); g.innerHTML='';
  opts.forEach(function(opt){
    var btn=document.createElement('button'); btn.className='c-btn'; btn.textContent=opt;
    if(opt===correctVal) btn.dataset.correct='1';
    (function(b,o){b.addEventListener('click',function(){
      g.querySelectorAll('.c-btn').forEach(function(x){x.disabled=true;});
      if(o===correctVal){b.className='c-btn ok';onCorrect(b);}
      else{b.className='c-btn ng';g.querySelectorAll('.c-btn').forEach(function(x){if(x.dataset.correct==='1')x.className='c-btn ok';});onWrong(b);}
    });})(btn,opt);
    g.appendChild(btn);
  });
}

// 用法をhintから判定
var USAGES=['副詞的用法・目的','副詞的用法・結果','副詞的用法・原因','副詞的用法・判断の根拠','形容詞的用法','名詞的用法'];
function getUsage(hint){
  if(!hint) return '副詞的用法・目的';
  if(hint.indexOf('結果')>=0) return '副詞的用法・結果';
  if(hint.indexOf('原因')>=0) return '副詞的用法・原因';
  if(hint.indexOf('判断')>=0) return '副詞的用法・判断の根拠';
  if(hint.indexOf('形容詞')>=0) return '形容詞的用法';
  if(hint.indexOf('名詞')>=0) return '名詞的用法';
  return '副詞的用法・目的';
}

function renderQ(){
  var c=deck[qi], tot=deck.length, isHard=(c.t==='hard');
  document.getElementById('q-num').textContent=isSudden?'連続:'+suddenScore+'問':'('+(qi+1)+'/'+tot+')';
  document.getElementById('q-pt').textContent=ep;
  document.getElementById('fb').textContent='';
  document.getElementById('fb').className='fb-area';
  document.getElementById('nxt-btn').style.display='none';
  document.getElementById('exp-box').style.display='none';

  if(isHard){
    // ===== 暗誦文モード：日本語を見て①英語を選ぶ ②用法を選ぶ =====
    document.getElementById('q-label').textContent='日本語 → 英語を選ぼう';
    document.getElementById('q-txt').textContent=c.a; // 日本語を表示
    document.getElementById('ja-choices-wrap').style.display='none';
    document.getElementById('en-choices-wrap').style.display='block';
    document.getElementById('usage-choices-wrap').style.display='none';
    document.getElementById('choices-label-en').textContent='📝 英語を選ぼう';

    var enWrongs=shuf(DATA.filter(function(x){return x.t==='hard'&&x.q!==c.q;})).slice(0,3).map(function(x){return x.q;});
    makeChoices('en-choices', shuf([c.q].concat(enWrongs)), c.q,
      function(){
        // 英語正解 → 用法選択へ
        setFb(true,'英語 OK！次は用法を選ぼう ↓');
        var correctUsage=getUsage(c.hint||'');
        var uWrongs=shuf(USAGES.filter(function(u){return u!==correctUsage;})).slice(0,3);
        document.getElementById('usage-choices-wrap').style.display='block';
        makeChoices('usage-choices', shuf([correctUsage].concat(uWrongs)), correctUsage,
          function(){
            var pt=PTS[lv]||30; ep+=pt; okCnt++;
            if(isSudden) suddenScore++;
            setFb(true,'+'+pt+' Pt　英語も用法も完璧！🎉');
            setKen('success','完璧！英語も用法もバッチリ！');
            showHint(c.hint);
            document.getElementById('nxt-btn').style.display='block';
          },
          function(){
            ngCnt++; addMistake(c.q);
            sessionMisses.push({q:c.q,type:'用法ミス'});
            setFb(false,'用法が違う… 正解: '+correctUsage);
            setKen('error','用法を確認しよう！');

            if(isSudden){showResult(true,'💀 用法ミスで即終了！連続: '+suddenScore+'問');return;}
            document.getElementById('nxt-btn').style.display='block';
          }
        );
      },
      function(){
        // 英語不正解
        ngCnt++; addMistake(c.q);
        sessionMisses.push({q:c.q,type:'英文ミス'});
        setFb(false,'不正解… 正解: '+c.q);
        setKen('error','惜しかった！解説を確認しよう。');
        showHint(c.hint);
        if(isSudden){showResult(true,'💀 不正解で即終了！連続: '+suddenScore+'問');return;}
        document.getElementById('nxt-btn').style.display='block';
      }
    );
    setKen('think','日本語を見て英語を選ぼう！');

  } else {
    // ===== 単語・フレーズモード：英語を見て日本語を選ぶ =====
    document.getElementById('q-label').textContent='英語 → 日本語を選ぼう';
    document.getElementById('q-txt').textContent=c.q; // 英語を表示
    document.getElementById('en-choices-wrap').style.display='none';
    document.getElementById('usage-choices-wrap').style.display='none';
    document.getElementById('ja-choices-wrap').style.display='block';

    var jaWrongs=shuf(DATA.filter(function(x){return x.a!==c.a;})).slice(0,3).map(function(x){return x.a;});
    makeChoices('ja-choices', shuf([c.a].concat(jaWrongs)), c.a,
      function(){
        var pt=PTS[lv]||PTS[c.t]||20; ep+=pt; okCnt++;
        if(isSudden) suddenScore++;
        setFb(true,'+'+pt+' Pt　正解！🎉');
        setKen('success','正解！さすがだ！');
        showHint(c.hint);
        document.getElementById('nxt-btn').style.display='block';
      },
      function(){
        ngCnt++; addMistake(c.q);
        sessionMisses.push({q:c.q,type:'選択ミス'});
        setFb(false,'不正解… 正解: '+c.a);
        setKen('error','惜しかった！解説を確認しよう。');
        showHint(c.hint);
        if(isSudden){showResult(true,'💀 不正解で即終了！連続: '+suddenScore+'問');return;}
        document.getElementById('nxt-btn').style.display='block';
      }
    );
    setKen('think','日本語訳を選んでみよう！');
  }
  startTimer();
}

      d.innerHTML='・'+m.q+'<span class="tag-f">'+m.type+'</span>';
      mlist.appendChild(d);
    });
  } else mbox.style.display='none';
  show('result-screen');
  updateUI();
  if(pct>=80&&!isOver) triggerCelebration();
  if(lv==='review'){
    var wrongQs=sessionMisses.map(function(m){return m.q;});
    deck.forEach(function(d){if(wrongQs.indexOf(d.q)<0) removeMistake(d.q);});
  }
}

function triggerCelebration(){
  var emojis=['✨','🌟','💎','🔥','📚','🎉','👑'];
  var colors=['#00e5ff','#ffc400','#00e676','#ff3d00','#ffffff'];
  var c=document.getElementById('particles'); c.innerHTML='';
  for(var i=0;i<70;i++){
    var p=document.createElement('div'); p.className='particle';
    p.style.left=(Math.random()*100)+'%';
    if(Math.random()>.3){p.style.width=(Math.random()*10+5)+'px';p.style.height=p.style.width;p.style.backgroundColor=colors[0|Math.random()*colors.length];if(Math.random()>.5)p.style.borderRadius='50%';}
    else{p.innerText=emojis[0|Math.random()*emojis.length];p.style.fontSize=(Math.random()*15+15)+'px';}
    p.style.animation='fallRotate '+(Math.random()*2+1.5)+'s cubic-bezier(.25,.46,.45,.94) forwards';
    p.style.setProperty('--tx',(Math.random()*300-150)+'px');
    p.style.setProperty('--rot',(Math.random()*720-360)+'deg');
    p.style.animationDelay=(Math.random()*.5)+'s';
    c.appendChild(p);
  }
  setTimeout(function(){c.innerHTML='';},4000);
}
